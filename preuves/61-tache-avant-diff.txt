+import { Booking } from "../store.js";
+import { overlaps } from "./overlap.js";
+
+export interface TimeSlot {
+  startsAt: string;
+  endsAt: string;
+}
+
+/**
+ * Retourne les creneaux libres d'une salle pour une journee donnee.
+ * Les creneaux sont horaires (00:00-01:00, 01:00-02:00, ..., 23:00-24:00).
+ */
+export function freeSlotsForDay(
+  bookings: Booking[],
+  date: string
+): TimeSlot[] {
+  const dayStart = new Date(`${date}T00:00:00Z`);
+  const dayEnd = new Date(`${date}T24:00:00Z`);
+
+  const dayBookings = bookings.filter((b) =>
+    overlaps(b.startsAt, b.endsAt, dayStart.toISOString(), dayEnd.toISOString())
+  );
+
+  const slots: TimeSlot[] = [];
+  for (let hour = 0; hour < 24; hour++) {
+    const slotStart = new Date(dayStart);
+    slotStart.setUTCHours(hour, 0, 0, 0);
+    const slotEnd = new Date(slotStart);
+    slotEnd.setUTCHours(hour + 1, 0, 0, 0);
+
+    const slotStartStr = slotStart.toISOString();
+    const slotEndStr = slotEnd.toISOString();
+
+    const isBooked = dayBookings.some((b) =>
+      overlaps(slotStartStr, slotEndStr, b.startsAt, b.endsAt)
+    );
+
+    if (!isBooked) {
+      slots.push({ startsAt: slotStartStr, endsAt: slotEndStr });
+    }
+  }
+
+  return slots;
+}
+
+export default freeSlotsForDay;

+  if (!room) {
+    res.status(404).json({ error: "salle inconnue" });

+  }
+

+    const slots = freeSlotsForDay(bookingsForRoom(room.id), date);
+    res.json({ roomId: room.id, date, slots });
+  } catch (err) {
+    if (err instanceof Error && "status" in err) {
+      res.status((err as { status: number }).status).json({ error: err.message });
+      return;
+    }
+    throw err;
+  }

+import { describe, it, expect } from "vitest";
+import { createApp } from "../src/app.js";
+
+interface ApiResponse {
+  status: number;
+  body: {
+    error?: string;
+    roomId?: string;
+    date?: string;
+    slots?: Array<{ startsAt: string; endsAt: string }>;
+  };
+}
+
+async function call(path: string, init?: RequestInit): Promise<ApiResponse> {
+  const app = createApp();
+  const server = app.listen(0);
+  const { port } = server.address() as { port: number };
+  try {
+    const res = await fetch(`http://127.0.0.1:${port}${path}`, init);
+    return { status: res.status, body: await res.json() };
+  } finally {
+    server.close();
+  }
+}
+
